# Pi5 wired-CarPlay test (protocol = carplay-wired). No Wi-Fi / Bluetooth / ESP32:
# the app brings the USB-plugged iPhone to config 6, opens com.apple.carkit.service
# via cp_usbmux + libimobiledevice, and the phone streams CarPlay over the USB-NCM
# link (usb0) to the :7000 server. Build with USE_CP_WIRED=1. Run as root (usbfs +
# i2c). renderer=none = headless log validation; use renderer=sdl to see the video.
protocol      = carplay-wired
mfi-i2c-bus   = /dev/i2c-1
mfi-i2c-addr  = 0x10
bluetooth-name = FastCarPlay
width         = 800
height        = 480
log-level     = 4
