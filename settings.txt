# FastCarPlay -- the full annotated settings reference.
#
# Every key the app knows, with its built-in default. A preset only needs the
# keys that differ (the device ships settings_drm.txt); changes made from the
# on-device UI land in ~/.fastcarplay/usersettings.txt, which is applied after
# the preset and survives image updates.
#
# Lines are `key = value`; `#` starts a comment.

# --- Source -----------------------------------------------------------------

# Protocol backend: "carplay-wireless", "carplay-wired", "aa-wireless" or
# "aa-usb" (native wired Android Auto via AOAP). Normally picked from the
# on-device UI (Settings > Source), which persists it and restarts.
#protocol = carplay-wireless

# --- Android Auto -----------------------------------------------------------

# Pin a specific phone for wired AA (0 = scan all devices for AOAP support).
#aa-vendor-id = 0
#aa-product-id = 0

# AA video mode: 1=800x480, 2=1280x720, 3=1920x1080 (the AA protocol only
# carries an enum of standard modes; CarPlay is asked for the panel's exact
# pixel size instead). `android-resolution` is the legacy alias.
#aa-resolution = 1
#aa-video-fps = 30

# UI scale reported to the phone.
#android-dpi = 120

# ms to wait for the phone to re-enumerate in accessory mode after the AOAP
# switch (first connections show a consent dialog on the phone).
#aa-accessory-timeout = 5000

# Media ack window offered in the AV setup response.
#aa-max-unacked = 4

# Offer only ChaCha20-Poly1305 for the AA TLS session. Much cheaper than
# AES-128-GCM on CPUs without AES acceleration (the ARM926 has none).
#force-chacha20 = false

# --- Wireless (Wi-Fi AP + Bluetooth) ----------------------------------------
# The access point is run by the system (init + hostapd), not by the app.
# wifi-ssid / wifi-passphrase / wifi-channel / wifi-ap-ip are MIRRORS of that
# configuration: they are read back from the effective hostapd.conf at
# start-up, and the Wireless screen writes changes into it. Setting them in a
# preset has no effect on the radio.
#wifi-interface = wlan0
#hostapd-conf = /etc/hostapd.conf
#ap-restart-command = /etc/init.d/S44ap stop; /etc/init.d/S44ap start
#wifi-ssid = FastCarPlay
#wifi-passphrase = carplay1234
#wifi-channel = 6
#wifi-ap-ip = 192.168.53.1
#bluetooth-name = FastCarPlay

# --- CarPlay ----------------------------------------------------------------

# The MFi 3.0 authentication coprocessor (wired and wireless CarPlay).
#mfi-i2c-bus = /dev/i2c-1
#mfi-i2c-addr = 16

# --- Display / video --------------------------------------------------------

# How video is decoded and shown: "auto" probes the kernel for a hardware
# decoder + DRM master; "hw" or "headless" force a path (see video_path.h).
#video-path = auto

# Headless-path geometry. On the DRM path the panel size comes from the
# kernel and these are ignored.
#width = 1024
#height = 576

# Frame rate requested from the phone (CarPlay); independent of the panel's
# refresh rate.
#source-fps = 60

# Frames the decoder may buffer before presentation.
#rendering-buffer = 5

# FFmpeg decoder tuning: low-delay mode and AV_CODEC_FLAG2_FAST (allows
# non-spec-compliant speedups).
#decode-low-delay = true
#decode-fast = true

# --- Touch ------------------------------------------------------------------

# evdev device for the touchscreen; empty = autodetect. The axis fixups cover
# panels mounted rotated.
#touch-device =
#touch-swap-xy = false
#touch-invert-x = false
#touch-invert-y = false

# --- On-device UI -----------------------------------------------------------

# Run the built-in screens (source picker, settings, wireless setup) on the
# overlay plane while no video plays. `lvgl-ui` is the legacy alias.
#ui = false

# --- USB (wired backends) ---------------------------------------------------

#async-usb-calls = 32
#usb-buffer-size = 2048
#usb-buffer = 128
#usb-write-timeout-ms = 300
#usb-write-retries = 6
#usb-write-retry-delay-ms = 2

# --- Audio ------------------------------------------------------------------

# ALSA PCM the in-app mixer opens (playback and capture).
#alsa-device = hw:0,0

# Output volume, 0-100 %. Settings > Audio and the volume keys of a Bluetooth
# keyboard/remote save it to ~/.fastcarplay/usersettings.txt, which overrides
# any value set here; -1 (the default) takes over the card's current level.
#volume = -1

# What sets the volume: auto (Master or PCM; else the card's only output amp,
# e.g. Headphone; else a shared digital control such as Digital, DAC or
# Playback; else a software gain), software, or the exact name of an ALSA
# mixer control (as alsamixer shows it).
#volume-control = auto

# ALSA control device for that; empty means the card of alsa-device.
#volume-device =

# Percent per volume key press or -/+ tap.
#volume-step = 5

# Buffers (in 10 ms periods) to hold before starting playback: music / call.
#audio-buffer-wait = 2
#audio-buffer-wait-call = 6

# Seconds of fade-in after a stream starts or ducks.
#audio-fade = 0.3

# Drop voice audio older than this instead of playing it late.
#audio-voice-max-latency-ms = 300

# Keep the PCM open this long after the last stream ends (avoids churn
# between back-to-back prompts).
#audio-idle-hold-ms = 1500

# Software gain (integer multiplier, saturating) applied to captured mic PCM.
# Raw speech already peaks near full scale on this board; raise only for a
# genuinely quiet mic fitting.
#mic-gain = 1

# Give the audio/video threads SCHED_FIFO priorities (needs root).
#realtime-priority = true

# --- System -----------------------------------------------------------------

# 0 none, 1 errors, 2 +warnings, 3 +info, 4 +verbose.
#log-level = 2

# Driving position; mirrors some UI layout the phone renders.
#left-hand-drive = true

# The phone's day/night look: 0 day, 1 night, 2 auto. Sent at connect and
# whenever it changes. Auto lets CarPlay follow the sun where the phone is;
# Android Auto has nothing of its own to follow and stays in day mode.
#night-mode = 2

# Run a script when a phone connects / disconnects.
#on-connect-script =
#on-disconnect-script =

# What Settings > System > Reboot / Shut down run, once the app has closed
# down cleanly. If one fails, the app starts again instead of leaving the
# screen dark.
#reboot-command = reboot
#shutdown-command = poweroff
