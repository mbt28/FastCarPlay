# Wireless CarPlay on the Raspberry Pi 5 dev box (software HEVC decode, SDL
# window on the labwc/Wayland desktop). The AP (hostapd ch36 5GHz + dnsmasq on
# wlan0 192.168.4.1) must already be running with these same credentials, and
# the MFi auth chip must be powered (GPIO4 high). Run: out/app settings_cp_pi.txt
protocol = carplay-wireless

wifi-interface  = wlan0
wifi-ssid       = FCP24test
wifi-passphrase = drive1234
wifi-channel    = 6
wifi-ap-ip      = 192.168.4.1
bluetooth-name  = FastCarPlay

window-mode = 0
width       = 800
height      = 480
source-fps  = 30
audio-driver = alsa
# The phone ships buffered media in bursts up to ~1s ahead; hold a matching
# cushion (~40 chunks ~= 920ms) so burst gaps + clock drift never starve SDL.
audio-buffer-wait = 40
log-level   = 5
