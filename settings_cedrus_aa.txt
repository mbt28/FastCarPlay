##############################################################################
# F1C200s native wired Android Auto profile (no Carlinkit dongle).
# Phone plugs straight into the board's USB host port; cedrus HW-decodes the
# H.264 and the DEFE scales it to the panel. Based on settings_cedrus.txt.
#
#   ./app settings_cedrus_aa.txt
#
# For first bring-up, watch the serial console: you should see
#   AOAP capable device ... -> Version handshake done -> TLS established
#   -> Service discovery request -> Phone connected (Android Auto)
##############################################################################

# 1. General ----------------------------------------------------------------

# Native wired Android Auto backend (AOAP).
protocol = aa-usb

# 0 = auto-scan the bus for an AOAP-capable phone. If auto-scan grabs the wrong
# device, set these to your phone's normal-mode VID/PID (decimals) from lsusb.
#aa-vendor-id = 0
#aa-product-id = 0

# Video the phone should stream. 1 = 800x480 @ 30fps: the lightest tier, the
# safe starting point for the arm926 (TLS + H.264 must fit in the CPU budget).
# The DEFE HW-scales it to the panel below, so this need not match the panel.
aa-resolution = 1
aa-video-fps = 30
# ms to wait for the phone to re-enumerate after the AOAP switch (first
# connection shows a consent dialog on the phone -- keep this generous).
aa-accessory-timeout = 8000
# Media ack window offered to the phone.
aa-max-unacked = 4
# Force the AA TLS session to ChaCha20-Poly1305. On the F1C200s (ARM926, no AES
# hardware) it roughly quartered the TLS-decrypt CPU vs AES-128-GCM. Set false
# if a phone won't connect (it then requires AES-GCM).
force-chacha20 = true

# Panel size (DEFE scales the 800x480 stream to this).
width = 480
height = 272
source-fps = 30

# Headless: decoder presents video on the DRM/DEFE plane, no window manager.
window-mode = 2

# Log level. 3 = info: shows the connect milestones above without the
# per-frame cost of level 4 (raise to 4 only to debug a stuck handshake).
log-level = 3

# 2. Device -----------------------------------------------------------------

# Standard USB current. If the phone brown-outs the board, keep this true.
weak-charge = true

# Driving position / night mode (sent to the phone as sensor data).
#left-hand-drive = true
#night-mode = 2

# Android Auto scale (DPI).
android-dpi = 135

# 3. Application ------------------------------------------------------------

# Mainline blob-free cedrus HW H.264 decode + DEFE display (same as the
# Carlinkit cedrus profile). The AA video channel is plain H.264, so this
# path is reused unchanged.
cedrus-decode = true
renderer = drm

# USB read pipeline tuning for the sunxi MUSB host controller. If you see
# "malformed"/desync or video lagging, raise async-usb-calls first.
async-usb-calls = 64
usb-buffer-size = 4096
usb-buffer = 256

# Audio output. "dummy" swallows audio (no sound) -- fine for the first video
# bring-up. To actually hear media/nav, set a real driver, e.g.:
#   audio-driver = alsa       (single stream; nav-over-music won't mix)
#   audio-driver = pulseaudio (if running on the board)
audio-driver = dummy

# 6. Debug ------------------------------------------------------------------

# Turn on to overlay frame/throughput info on the video (useful for judging
# whether the arm926 keeps up with the stream).
#debug-overlay = true
