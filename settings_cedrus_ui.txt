##############################################################################
# F1C200s native UI profile (LVGL on the DRM overlay plane).
#
# Same hardware path as settings_cedrus_aa.txt (cedrus HW decode, DEFE, DRM),
# but the on-device UI is the home screen: pick the source and change settings
# on the unit itself with the touchscreen (and a rotary encoder once wired).
#
#   ./app settings_cedrus_ui.txt      (on the board)
#
# The source you pick is written to /root/.fastcarplay/usersettings.txt and
# applied on the next start, so the `protocol` below is only the first-boot
# default. Deleting that file resets to this preset.
##############################################################################

# 1. UI ---------------------------------------------------------------------

# LVGL screens on the DRM overlay plane as the home screen. Touch drives it;
# selecting a source persists it and the app re-execs into that protocol.
lvgl-ui = true

# Icon set: material (Material Symbols) or builtin (LVGL's bundled symbols).
icon-theme = material

# 2. General ----------------------------------------------------------------

# First-boot default source; the UI overrides this once the user picks one.
protocol = aa-usb

# Native AA video stream: 1 = 800x480 @ 30fps (lightest tier for the arm926).
aa-resolution = 1
aa-video-fps = 30
aa-accessory-timeout = 8000
aa-max-unacked = 4
# ChaCha20-Poly1305 TLS: ~4x cheaper than AES-GCM on the ARM926 (no AES HW).
force-chacha20 = true

# Panel size. The UI lays itself out from these; the DEFE scales AA video.
width = 480
height = 272
source-fps = 30

# Headless DRM: video on the DEFE plane, UI on the ARGB overlay above it.
window-mode = 2
log-level = 3

# 3. Device -----------------------------------------------------------------

weak-charge = true
android-dpi = 135

# 4. Application ------------------------------------------------------------


# USB read pipeline tuning for the sunxi MUSB host controller.
async-usb-calls = 64
usb-buffer-size = 4096
usb-buffer = 256

# Control-channel write retry (MUSB DMA BUS_SEL collision recovery).
usb-write-timeout-ms = 300
usb-write-retries = 6
usb-write-retry-delay-ms = 2

audio-driver = dummy
