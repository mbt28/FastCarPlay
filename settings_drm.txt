# FastCarPlay -- head unit (the app owns the screen via DRM/KMS).
#
# For the F1C200s and any other board that boots without a desktop: video goes
# on a DRM plane, the LVGL UI on an overlay plane above it. Nothing here selects
# a decoder or a renderer -- video-path defaults to "auto", which asks the kernel
# what this chip can decode and picks hardware + DRM when it can, software
# otherwise (see video_path.h). The CarPlay codec offer follows from that too.
#
# The backend below is only the FIRST-BOOT default: pick another from the UI
# (Settings > Source), which saves it to usersettings.txt and restarts.
#
# Only values that differ from the built-in defaults are listed. See settings.txt
# for the full annotated list of options.

protocol   = carplay-wired
lvgl-ui    = true
icon-theme = material

# No panel size here: in DRM mode the app asks the kernel (connector/CRTC) and
# requests exactly that from the phone, so nothing has to be kept in sync with
# the panel. window-mode only applies if DRM is unavailable and we fall back to
# an SDL window. source-fps is what we ask the phone for and is independent of
# the panel's refresh rate -- 30fps content on a 60Hz panel is intentional.
window-mode = 2
source-fps  = 60
android-dpi = 135

# The HW decoder presents directly to the DRM plane, so the app-side frame
# buffer is only drained, never displayed: the minimum (3) is enough and
# frees two pinned decoder frames vs the default 5.
rendering-buffer = 3

audio-driver = dummy

# The ARM926 has no AES instructions, so ChaCha20-Poly1305 is far cheaper for
# the AA/CarPlay session ciphers than the AES-GCM a phone would otherwise pick.
force-chacha20 = true

# USB tuning for the F1C200s MUSB controller (deeper queues, bigger transfers,
# and a longer wait for a phone to re-enumerate in accessory mode).
async-usb-calls      = 64
usb-buffer           = 256
usb-buffer-size      = 4096
aa-accessory-timeout = 8000

mfi-i2c-bus   = /dev/i2c-0
mfi-i2c-addr  = 0x10

# Wi-Fi AP for the wireless backends (aa-wireless / carplay-wireless): NOT set
# here. The AP belongs to the system (hostapd + dnsmasq from /etc/hostapd.conf,
# started by S43wifi/S44ap over the ESP32), and the app reads its effective
# SSID / passphrase / channel / IP from that file at connect time -- so the
# credentials handed to the phone always describe the network that is actually
# up. Hardcoding wifi-ssid/wifi-passphrase/wifi-ap-ip here only re-introduces
# drift: change the AP in /etc/hostapd.conf instead.

log-level = 3
