# FastCarPlay -- desktop / development (the app runs in a window).
#
# For a dev box or a Pi running a desktop session. video-path defaults to "auto":
# it sees the compositor (WAYLAND_DISPLAY / DISPLAY), leaves KMS alone and draws
# into an SDL window with software decode. Nothing to configure for video.
#
# Switch backend from the UI (Settings > Source) if the LVGL UI is enabled, or
# just change protocol below. Wired CarPlay additionally needs the one-time host
# setup -- see docs/wired-carplay.md and scripts/setup-wired-carplay.sh.
#
# Only values that differ from the built-in defaults are listed. See settings.txt
# for the full annotated list of options.

protocol = carplay-wired
lvgl-ui    = true
icon-theme = material

# A window the size of a typical head-unit panel, so the UI is laid out as it
# will be on the device.
width       = 800
height      = 480
source-fps  = 60
window-mode = 0

audio-driver = alsa

# Wi-Fi AP for the wireless backends (aa-wireless / carplay-wireless): NOT set
# here. The app reads the effective SSID / passphrase / channel / IP from the
# system AP's /etc/hostapd.conf at connect time, so the credentials handed to
# the phone always match the network that is actually up. Point hostapd-conf at
# a real hostapd config (with hostapd + dnsmasq running) to use a desktop AP.

log-level = 4
